How it works

def recommend(df, book, n=5):
    ids = df[df.Book_Title == book].Student_ID.unique()
    others = df[df.Student_ID.isin(ids)
                & (df.Book_Title != book)]
    return (others.groupby('Book_Title')
            .Student_ID.nunique().nlargest(n))
